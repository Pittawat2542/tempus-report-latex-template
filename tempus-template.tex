\subsection{Algorithms}
\Cref{alg:mean} describes the sample mean from \cref{eq:mean}.
Algorithms use a floating caption, optional line numbers, and themed
keywords. Place the label immediately after the caption.

\begin{algorithm}[htbp]
  \caption{Compute the sample mean.}
  \label{alg:mean}
  \begin{algorithmic}[1]
    \Require Measurements $x_1,\ldots,x_n$ with $n > 0$
    \Ensure Sample mean $\bar{x}$
    \State $s \gets 0$ \Comment{Running total}
    \For{$i \gets 1$ \textbf{to} $n$}
      \State $s \gets s + x_i$
    \EndFor
    \State \Return $s/n$
  \end{algorithmic}
\end{algorithm}

\subsection{Code listings}
The same calculation appears in \cref{lst:mean}. Listings provide syntax
colors, line numbers, and automatic line wrapping without shell escape.

\begin{lstlisting}[language=Python,caption={A minimal sample mean implementation.},label={lst:mean}]
def sample_mean(values):
    """Return the arithmetic mean of a nonempty sequence."""
    if not values:
        raise ValueError("At least one measurement is required")
    # Keep the example close to the pseudocode.
    total = sum(values)
    return total / len(values)
\end{lstlisting}
